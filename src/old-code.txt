String getGPSCoordinates2() {
  String lat = "";
  String lng = "";
  //String tm = "";
  bool fixAcquired = false;
  String gpsResponse;
  
  Serial2.println("ATE0");
  // Turn on the GPS
  sendATCommand("AT+CGNSPWR=1", "OK", 2000);
  delay(2000); // Give it some time to start up

  while (!fixAcquired) {
    // Get GPS data
    gpsResponse = sendATCommand("AT+CGNSINF", "", 2000);
    Serial.println(gpsResponse);
    // Check if GPS has a fix
    if (gpsResponse.indexOf("+CGNSINF: 1") != -1) {
      // Extract latitude, longitude, and altitude
      int firstComma = gpsResponse.indexOf(',', 9); // Skip "+CGNSINF: "
      int secondComma = gpsResponse.indexOf(',', firstComma + 1);
      int thirdComma = gpsResponse.indexOf(',', secondComma + 1);
      int fourthComma = gpsResponse.indexOf(',', thirdComma + 1);
      //tm = gpsResponse.substring(secondComma + 1, thirdComma);

      lat = gpsResponse.substring(thirdComma + 1, fourthComma);
      int fifthComma = gpsResponse.indexOf(',', fourthComma + 1);
      lng = gpsResponse.substring(fourthComma + 1, fifthComma);

      // Check if we have valid lat/lng data
      if (lat != "" && lng != "") {
        fixAcquired = true;
        sim808Location.longitude = lng.toFloat();
        sim808Location.latitude = lat.toFloat();
        Serial.println("Fix aquired");
        Serial.println("Latitude: " + lat);
        Serial.println("Longitude: " + lng);
      } else {
        Serial.println("No valid latitude, longitude, or altitude.");
        delay(5000); // Wait for 5 seconds before checking again
      }
    } else {
      Serial.println("No GPS fix yet. Waiting for GPS fix...");
      delay(5000); // Wait for 5 seconds before checking again
    }
  }
  // Turn on the GPS
  sendATCommand("AT+CGNSPWR=0", "OK", 2000);  // Turn off GPS
  return "Lat: " + lat + ", Lng: " + lng;
}
